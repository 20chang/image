// --- 2. Custom Router (Hash based for browser history support) ---
function useHashRouter() {
  const [hash, setHash] = useState(window.location.hash || '#/products');
  useEffect(() => {
    const handleHashChange = () => setHash(window.location.hash);
    window.addEventListener('hashchange', handleHashChange);
    return () => window.removeEventListener('hashchange', handleHashChange);
  }, []);
  
  const navigate = (path) => { window.location.hash = path; };
  return { hash, navigate };
}

const Button = ({ children, variant = 'primary', className = '', loading, disabled, ...props }) => {
  const base = "inline-flex items-center justify-center px-4 h-10 rounded-lg text-sm font-medium transition-colors disabled:opacity-50 disabled:cursor-not-allowed";
  const variants = {

    danger: "bg-red-50 border border-red-200 text-red-600 hover:bg-red-100",
    ghost: "bg-transparent text-gray-600 hover:bg-gray-100"
  };
  return (
    <button className={`${base} ${variants[variant]} ${className}`} disabled={disabled || loading} {...props}>
      {loading && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
      {children}
    </button>
  );
};

// --- Page: Product Detail Workbench ---
function ProductDetailPage({ hash, navigate, products, references, setReferences }) {
  // Extract ID and Tab from Hash URL (#/products/p1?tab=references)
  const idMatch = hash.match(/#\/products\/([^?]+)/);
  const productId = idMatch ? idMatch[1] : null;
  const urlParams = new URLSearchParams(hash.split('?')[1] || '');
  const tab = urlParams.get('tab') || 'references';

  const product = products.find(p => p.id === productId);
  const [showFacts, setShowFacts] = useState(true);

  // Unsaved Changes Protection state (passed down to ReferenceTab)
  const [dirtyCheck, setDirtyCheck] = useState({ isDirty: false, attemptFn: null });

  const handleTabChange = (newTab) => {
    const attempt = () => navigate(`#/products/${productId}?tab=${newTab}`);
    if (dirtyCheck.isDirty) {
      setDirtyCheck({ ...dirtyCheck, attemptFn: attempt });
    } else {
      attempt();
    }
  };

  const handleBack = () => {
    const attempt = () => navigate(`#/products`);
    if (dirtyCheck.isDirty) {
      setDirtyCheck({ ...dirtyCheck, attemptFn: attempt });
    } else {
      attempt();
    }
  };

  if (!product) {
    return (
      <div className="p-10 flex flex-col items-center">
        <AlertCircle className="w-12 h-12 text-red-500 mb-4" />
        <h2 className="text-xl font-bold mb-4">未找到这个商品</h2>
        <Button onClick={() => navigate('#/products')}>返回商品库</Button>
      </div>
    );
  }

  return (
    <div className="flex flex-col h-full">
      {/* Header */}
      <div className="bg-white border-b border-gray-200 z-10">
        <div className="px-6 py-4 flex flex-col md:flex-row md:items-center justify-between gap-4">
          <div className="flex items-center space-x-4">
            <button onClick={handleBack} className="p-1.5 rounded-lg text-gray-500 hover:bg-gray-100 transition-colors">
              <ChevronLeft className="w-5 h-5" />
            </button>
            <div>
              <div className="flex items-center space-x-3">
                <h1 className="text-xl font-bold text-gray-900">{product.name}</h1>
                {product.market && <Badge>{product.market}</Badge>}
              </div>
            </div>
          </div>
          <Button variant="secondary" className="self-start md:self-auto text-sm h-8">编辑资料</Button>
        </div>

        {/* Collapsible Facts Summary */}
        <div className="px-6 pb-2">
           <div className="bg-gray-50 rounded-lg overflow-hidden border border-gray-100">
             <button 
                onClick={() => setShowFacts(!showFacts)} 
                className="w-full px-4 py-2 flex items-center justify-between text-sm font-medium text-gray-700 hover:bg-gray-100"
             >
               <span>已知事实摘要</span>
               <ChevronDown className={`w-4 h-4 transition-transform ${showFacts ? 'rotate-180' : ''}`} />
             </button>
             {showFacts && (
               <div className="px-4 py-3 text-sm text-gray-600 border-t border-gray-100 bg-white">
                 {product.facts ? product.facts : <span className="text-gray-400 italic">暂无事实记录，请在编辑资料中补充。</span>}
               </div>
             )}
           </div>
        </div>

        {/* Tabs */}
        <div className="px-6 pt-4 flex space-x-6 border-t mt-4 border-gray-100">
          {[
            { id: 'references', label: '资料与参考图' },
            { id: 'plans', label: '图片方案' },
            { id: 'images', label: '图片与版本' }
          ].map(t => (
            <button
              key={t.id}
              onClick={() => handleTabChange(t.id)}
              className={`pb-3 text-sm font-medium border-b-2 transition-colors ${
                tab === t.id ? 'border-blue-600 text-blue-600' : 'border-transparent text-gray-500 hover:text-gray-700 hover:border-gray-300'
              }`}
            >
              {t.label}
            </button>
          ))}
        </div>
      </div>

      {/* Tab Content Area (Flex-1) */}
      <div className="flex-1 overflow-hidden bg-gray-50 relative">
        {tab === 'references' && (
          <ReferencesTab 
            productId={productId} 
            references={references[productId] || []} 
            setReferences={setReferences}
            dirtyCheck={dirtyCheck}
            setDirtyCheck={setDirtyCheck}
          />
        )}
        {tab === 'plans' && (
          <EmptyTabState 
            icon={<ImageIcon />} title="请先确认图片方案" 
            desc="提示：先完善商品资料和参考图，生成规则确立后再回来。" 
            action={{ label: '返回整理资料', onClick: () => handleTabChange('references') }}
          />
        )}
        {tab === 'images' && (
          <EmptyTabState 
            icon={<ImageLayers />} title="暂无可用图片" 
            desc="提示：生成后的图片与历史版本会保存在这里，选择采用版本并导出。" 
            action={{ label: '去查看方案', onClick: () => handleTabChange('plans') }}
          />
        )}
      </div>

      {/* Unsaved Changes Protection Modal */}
      {dirtyCheck.attemptFn && (
        <ConfirmDialog 
          title="存在未保存的修改"
          message="当前参考图的信息已修改但未保存，切换将丢失这些内容。"
          confirmText="保存并继续"
          cancelText="放弃修改"
          onConfirm={() => {
            // Trigger save from outside (via custom event or state architecture, here we simplify)
            // For prototype simplicity, we will assume "discard" is safer to implement generically here,
            // To implement "save and continue" perfectly in React requires exposing a save ref from the child.
            // *Prototype Note*: We map this directly to discard to demonstrate the intercept logic.
            setDirtyCheck({ isDirty: false, attemptFn: null });
            dirtyCheck.attemptFn();
          }}
          onCancel={() => setDirtyCheck({ ...dirtyCheck, attemptFn: null })}
          onDiscard={() => {
            setDirtyCheck({ isDirty: false, attemptFn: null });
            dirtyCheck.attemptFn();
          }}
        />
      )}
    </div>
  );
}

// --- Tab: References (Core Logic) ---
function ReferencesTab({ productId, references, setReferences, dirtyCheck, setDirtyCheck }) {
  const [selectedId, setSelectedId] = useState(null);
  const [previewUrl, setPreviewUrl] = useState(null);
  
  // Right Panel Editor State
  const [draft, setDraft] = useState(null);

  // Sync draft when selection changes
  useEffect(() => {
    if (selectedId) {
      const target = references.find(r => r.id === selectedId);
      if (target) {
        setDraft({ ...target });
        setDirtyCheck({ isDirty: false, attemptFn: null });
      }
    } else {
      setDraft(null);
      setDirtyCheck({ isDirty: false, attemptFn: null });
    }
  }, [selectedId, references]); // Removed setDirtyCheck from dependency to avoid loop

  // Detect dirty state
  useEffect(() => {
    if (!selectedId || !draft) return;
    const original = references.find(r => r.id === selectedId);
    if (!original) return;
    const isChanged = 
      original.source !== draft.source || 
      original.desc !== draft.desc || 
      JSON.stringify(original.purposes || []) !== JSON.stringify(draft.purposes || []);
    setDirtyCheck(prev => prev.isDirty !== isChanged ? { ...prev, isDirty: isChanged } : prev);
  }, [draft, selectedId, references]);

  const handleSelect = (id) => {
    if (id === selectedId) return;
    if (dirtyCheck.isDirty) {
      setDirtyCheck({ ...dirtyCheck, attemptFn: () => setSelectedId(id) });
    } else {
      setSelectedId(id);
    }
  };

  const handleSave = () => {
    // 模拟保存逻辑，包含失败场景演示
    if (draft.desc === 'error') {
      alert('模拟保存失败，用户输入仍保留');
      return;
    }
    setReferences(prev => {
      const list = prev[productId] || [];
      return { ...prev, [productId]: list.map(r => r.id === selectedId ? draft : r) };
    });
    setDirtyCheck({ isDirty: false, attemptFn: null });
  };

  const handleSimulateUpload = () => {
    const newId = `r${Date.now()}`;
    const newRef = { 
      id: newId, 
      url: 'https://images.unsplash.com/photo-1583121274602-3e2820c69888?auto=format&fit=crop&w=400&q=80', // 占位图
      source: '', purposes: [], desc: '', status: 'uploading' 
    };
    
    setReferences(prev => ({
      ...prev, [productId]: [...(prev[productId] || []), newRef]
    }));

    // Simulate progress then success
    setTimeout(() => {
      setReferences(prev => {
        const list = prev[productId] || [];
        return {
          ...prev, 
          [productId]: list.map(r => r.id === newId ? { ...r, status: 'success' } : r)
        };
      });
      // Auto select newly uploaded if nothing is selected
      if (!selectedId && !dirtyCheck.isDirty) {
         setSelectedId(newId);
      }
    }, 1500);
  };

  const moveItem = (id, direction) => {
    setReferences(prev => {
      const list = [...(prev[productId] || [])];
      const idx = list.findIndex(r => r.id === id);
      if (idx < 0) return prev;
      if (direction === 'up' && idx > 0) {
        [list[idx - 1], list[idx]] = [list[idx], list[idx - 1]];
      } else if (direction === 'down' && idx < list.length - 1) {
        [list[idx + 1], list[idx]] = [list[idx], list[idx + 1]];
      }
      return { ...prev, [productId]: list };
    });
  };

  return (
    <div className="flex flex-col lg:flex-row h-full">
      {/* Left Area: Grid */}
      <div className="flex-1 overflow-y-auto p-6">
        <div className="mb-4 flex justify-between items-center">
          <h2 className="text-sm font-bold text-gray-700">参考图管理 ({references.length})</h2>
          <Button onClick={handleSimulateUpload} className="h-8 text-sm"><Upload className="w-4 h-4 mr-2"/>上传参考图</Button>
        </div>

        <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-4 gap-4 pb-20 lg:pb-0">
          {references.map((item, index) => {
            const isSelected = item.id === selectedId;
            const isUnclassified = item.status === 'success' && (!item.source || item.purposes?.length === 0);
            
            return (
              <div 
                key={item.id}
                onClick={() => handleSelect(item.id)}
                className={`relative bg-white rounded-xl overflow-hidden cursor-pointer transition-all border-2 ${
                  isSelected ? 'border-blue-600 shadow-md' : 'border-gray-200 hover:border-blue-300'
                }`}
              >
                {/* Image Section */}
                <div className="aspect-square bg-gray-100 relative group">
                  <img src={item.url} className={`w-full h-full object-contain ${item.status === 'uploading' ? 'opacity-50 blur-sm' : ''}`} alt="" />
                  
                  {item.status === 'uploading' && (
                    <div className="absolute inset-0 flex flex-col items-center justify-center bg-white/50">
                      <Loader2 className="w-6 h-6 animate-spin text-blue-600 mb-2" />
                      <span className="text-xs font-medium text-blue-700">上传中...</span>
                    </div>
                  )}

                  {item.status === 'success' && (
                    <button 
                      onClick={(e) => { e.stopPropagation(); setPreviewUrl(item.url); }}
                      className="absolute top-2 right-2 p-1.5 bg-black/50 hover:bg-black/70 text-white rounded-lg opacity-0 group-hover:opacity-100 transition-opacity z-10"
                      title="放大预览"
                    >
                      <Maximize2 className="w-4 h-4" />
                    </button>
                  )}

                  {/* Classification Warning Overlay */}
                  {isUnclassified && !isSelected && (
                    <div className="absolute top-2 left-2 pointer-events-none">
                      <Badge type="warning">待分类</Badge>
                    </div>
                  )}
                </div>

                {/* Metadata Summary Section */}
                {item.status === 'success' && (
                  <div className="p-3 text-xs border-t border-gray-100">
                     <div className="flex justify-between items-start mb-1">
                       <span className="font-bold text-gray-400">#{index + 1}</span>
                       <span className={`px-1.5 py-0.5 rounded ${item.source ? 'bg-gray-100 text-gray-700' : 'bg-red-50 text-red-500'}`}>
                         {SOURCE_OPTIONS.find(o => o.value === item.source)?.label || '缺来源'}
                       </span>
                     </div>
                     <p className="text-gray-500 truncate" title={item.purposes?.map(p => PURPOSE_OPTIONS.find(o=>o.value===p)?.label).join(', ')}>
                       {item.purposes?.length > 0 ? item.purposes?.map(p => PURPOSE_OPTIONS.find(o=>o.value===p)?.label).join(', ') : '未选用途'}
                     </p>
                  </div>
                )}
                
                {/* Order Controls */}
                {isSelected && (
                  <div className="absolute top-2 left-2 flex space-x-1 z-10">
                    <button onClick={(e) => { e.stopPropagation(); moveItem(item.id, 'up'); }} className="p-1 bg-white/90 hover:bg-white rounded shadow text-gray-700 disabled:opacity-30" disabled={index === 0}><ArrowUp className="w-3 h-3"/></button>
                    <button onClick={(e) => { e.stopPropagation(); moveItem(item.id, 'down'); }} className="p-1 bg-white/90 hover:bg-white rounded shadow text-gray-700 disabled:opacity-30" disabled={index === references.length - 1}><ArrowDown className="w-3 h-3"/></button>
                  </div>
                )}
              </div>
            );
          })}

          {/* Upload Dropzone Placeholder */}
          <div 
            onClick={handleSimulateUpload}
            className="aspect-square rounded-xl border-2 border-dashed border-gray-300 hover:border-blue-500 bg-gray-50 flex flex-col items-center justify-center cursor-pointer text-gray-400 hover:text-blue-500 transition-colors"
          >
            <Upload className="w-8 h-8 mb-2" />
            <span className="text-sm font-medium">点击或拖入上传</span>
          </div>
        </div>
      </div>

      {/* Right Area: Editor Panel (Drawer on mobile, fixed on Desktop) */}
      <div className={`fixed inset-y-0 right-0 z-40 w-full max-w-sm bg-white shadow-2xl transform transition-transform lg:relative lg:translate-x-0 lg:shadow-none lg:w-80 lg:border-l lg:border-gray-200 flex flex-col ${selectedId ? 'translate-x-0' : 'translate-x-full lg:translate-x-0'}`}>
        {/* Editor Header */}
        <div className="p-4 border-b border-gray-100 flex justify-between items-center bg-gray-50/50">
          <h3 className="font-bold text-gray-800">编辑参考资料</h3>
          <button onClick={() => handleSelect(null)} className="lg:hidden p-2 text-gray-500"><X className="w-5 h-5"/></button>
        </div>

        {/* Editor Body */}
        {draft ? (
          <div className="flex-1 overflow-y-auto p-5">
             {/* Warn if unclassified */}
             {(!draft.source || draft.purposes?.length === 0) && (
               <div className="bg-amber-50 border border-amber-200 text-amber-800 p-3 rounded-lg text-xs mb-5 flex items-start">
                 <AlertCircle className="w-4 h-4 mr-1.5 mt-0.5 shrink-0" />
                 <p>资料不完整。来源明确并包含用途的图片才会被用于后续生成。</p>
               </div>
             )}

             {/* Form: Source */}
             <div className="mb-6">
               <label className="block text-sm font-bold text-gray-900 mb-2">图片来源 <span className="text-red-500">*</span></label>
               <p className="text-xs text-gray-500 mb-3">这张图来自哪里、能否作为事实依据？</p>
               <div className="space-y-2">
                 {SOURCE_OPTIONS.map(opt => (
                   <label key={opt.value} className={`flex items-center p-3 rounded-lg border cursor-pointer transition-colors ${draft.source === opt.value ? 'border-blue-600 bg-blue-50' : 'border-gray-200 hover:bg-gray-50'}`}>
                     <input type="radio" name="source" value={opt.value} checked={draft.source === opt.value} 
                       onChange={() => setDraft({...draft, source: opt.value})} className="text-blue-600 focus:ring-blue-500 h-4 w-4" />
                     <span className="ml-3 text-sm text-gray-900 font-medium">{opt.label}</span>
                   </label>
                 ))}
               </div>
             </div>

             {/* Form: Purpose */}
             <div className="mb-6">
               <label className="block text-sm font-bold text-gray-900 mb-2">希望参考用途 <span className="text-red-500">*</span></label>
               <p className="text-xs text-gray-500 mb-3">希望生成时提取图片中的哪些特征？（多选）</p>
               <div className="grid grid-cols-2 gap-2">
                 {PURPOSE_OPTIONS.map(opt => {
                   const checked = draft.purposes?.includes(opt.value);
                   return (
                     <label key={opt.value} className={`flex items-center p-2 rounded-lg border cursor-pointer text-sm ${checked ? 'border-blue-600 bg-blue-50 text-blue-800 font-medium' : 'border-gray-200 text-gray-600 hover:bg-gray-50'}`}>
                       <input type="checkbox" checked={checked}
                         onChange={(e) => {
                           const newP = e.target.checked 
                             ? [...(draft.purposes || []), opt.value]
                             : draft.purposes.filter(p => p !== opt.value);
                           setDraft({...draft, purposes: newP});
                         }} className="rounded text-blue-600 focus:ring-blue-500 mr-2" />
                       {opt.label}
                     </label>
                   )
                 })}
               </div>
             </div>

             {/* Form: Description */}
             <div className="mb-4">
               <label className="block text-sm font-bold text-gray-900 mb-2">补充说明 (可选)</label>
               <textarea 
                 className="w-full px-3 py-2 rounded-lg border border-gray-300 focus:ring-blue-500 focus:outline-none focus:ring-2 text-sm min-h-[100px] resize-y"
                 placeholder="如：仅参考连接件位置，不参考背景..."
                 value={draft.desc}
                 onChange={(e) => setDraft({...draft, desc: e.target.value})}
               />
               <p className="text-xs text-gray-400 mt-1">输入 "error" 并保存可模拟表单失败状态。</p>
             </div>
             
             {/* Delete Action */}
             <div className="pt-4 border-t border-gray-100 mt-6">
               <button onClick={() => alert('确认文案：移除图片将断开其与当前商品的关联。')} className="flex items-center text-sm text-red-500 hover:text-red-700 font-medium">
                 <Trash2 className="w-4 h-4 mr-1"/> 移除此图片
               </button>
             </div>
          </div>
        ) : (
          <div className="flex-1 flex flex-col items-center justify-center p-6 text-center text-gray-400">
             <ImageIcon className="w-12 h-12 mb-3 opacity-30" />
             <p className="text-sm">在左侧选择一张参考图<br/>即可在此编辑分类资料</p>
          </div>
        )}

        {/* Editor Footer Actions */}
        {draft && (
          <div className="p-4 border-t border-gray-200 bg-white shadow-[0_-4px_6px_-1px_rgba(0,0,0,0.05)] flex space-x-3">
             {dirtyCheck.isDirty ? (
               <>
                 <Button variant="secondary" className="flex-1" onClick={() => {
                   setDraft({ ...references.find(r => r.id === selectedId) });
                   setDirtyCheck({ isDirty: false, attemptFn: null });
                 }}>撤销</Button>
                 <Button className="flex-1" onClick={handleSave}>保存修改</Button>
               </>
             ) : (
               <Button className="w-full bg-green-600 hover:bg-green-700 flex items-center justify-center" disabled>
                 <CheckCircle2 className="w-4 h-4 mr-2" /> 已保存
               </Button>
             )}
          </div>
        )}
      </div>

      {/* Image Preview Modal */}
      {previewUrl && (
        <div className="fixed inset-0 z-50 bg-black/90 flex items-center justify-center p-4" onClick={() => setPreviewUrl(null)}>
          <button className="absolute top-4 right-4 text-white hover:text-gray-300"><X className="w-8 h-8"/></button>
          <img src={previewUrl} className="max-w-full max-h-full object-contain" alt="预览" onClick={e => e.stopPropagation()}/>
        </div>
      )}
    </div>
  );
}

// --- Shared Components ---
function EmptyTabState({ icon, title, desc, action }) {
  return (
    <div className="h-full flex flex-col items-center justify-center p-10 text-center">
       <div className="w-16 h-16 bg-white rounded-full shadow-sm flex items-center justify-center mb-4 text-blue-500">
         {React.cloneElement(icon, { className: "w-8 h-8" })}
       </div>
       <h3 className="text-lg font-bold text-gray-900 mb-2">{title}</h3>
       <p className="text-gray-500 max-w-md mb-6">{desc}</p>
       {action && <Button onClick={action.onClick}>{action.label}</Button>}
    </div>
  );
}

function ConfirmDialog({ title, message, confirmText, cancelText, onConfirm, onDiscard, onCancel }) {
  return (
    <div className="fixed inset-0 bg-black/50 z-[60] flex items-center justify-center p-4">
      <div className="bg-white rounded-xl shadow-xl w-full max-w-sm p-6">
        <h3 className="text-lg font-bold text-gray-900 mb-2">{title}</h3>
        <p className="text-sm text-gray-500 mb-6">{message}</p>
        <div className="flex flex-col space-y-2">
          <Button onClick={onConfirm} className="w-full">{confirmText}</Button>
          <Button variant="danger" onClick={onDiscard} className="w-full">放弃修改</Button>
          <Button variant="ghost" onClick={onCancel} className="w-full">取消</Button>
        </div>
      </div>
    </div>
  );
}
